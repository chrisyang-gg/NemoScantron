These rules evaluate the velocity_features block and card/account context to
detect behavioral anomalies. Each rule's risk_contribution is a raw score
added to velocity_behavioral_score before section weighting is applied.

RULE VB-001 | HIGH TRANSACTION FREQUENCY — LAST HOUR
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  rule_id             : VB-001
  condition           : velocity_features.transactions_last_1h >= 5
  risk_contribution   : 0.40
  severity            : high
  notes               : Five or more transactions in one hour is outside the
                        norm for most cardholders. Escalate weighting by +0.10
                        for every additional transaction beyond 5.
                        Cross-reference with VB-005 for compounding effect.

RULE VB-002 | LARGE AMOUNT DEVIATION FROM PERSONAL BASELINE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  rule_id             : VB-002
  condition           : amount > (velocity_features.avg_transaction_amount_90d
                          + (3.0 × velocity_features.std_transaction_amount_90d))
  risk_contribution   : 0.30
  severity            : medium
  notes               : Three-standard-deviation outlier from this cardholder's
                        own 90-day baseline. Weight increases by +0.10 if
                        channel is card_not_present or online.
