{
  "transaction_id"      : <string>  Copied verbatim from the input record.

  "analysis_timestamp"  : <string>  ISO 8601 UTC datetime of this analysis,
                                    e.g. "2026-09-19T14:32:07Z".

  "risk_score"          : <number>  Composite fraud risk score.
                                    Range: 0.000 to 1.000. Three decimal places.

                                      "approve"
                                      "flag_for_review"
                                      "hold"
                                      "decline"
                                    No other values are permitted.

  "confidence"          : <number>  Confidence in the risk_score given the
                                    completeness of the input data.
                                    Range: 0.000 to 1.000. Three decimal places.

  "rules_triggered"     : <array>   Every rule_id (e.g. "VB-001", "GEO-003")
                                    that fired during this analysis, as strings.
                                    Empty array [] if no rules fired.

  "patterns_matched"    : <array>   Every pattern_id (e.g. "CARD-TESTING") that
                                    met its detection threshold. Empty array []
                                    if no patterns matched.

  "fraud_indicators"    : <array>   One entry per material signal observed.
                                    Required: at least one entry whenever
                                    decision is anything other than "approve".
                                    May be empty [] for clean approvals only.
                                    Each entry:
                                    {
                                      "indicator" : <string> Short signal name.
                                      "severity"  : <string> Exactly one of:
                                                      "low"
                                                      "medium"
                                                      "high"
                                                      "critical"
                                      "detail"    : <string> One sentence
                                                    describing what was observed.
                                    }

                                    weighting and pattern boost.

    "geo_device"         : <number> Range: 0.000–1.000.
    "merchant_auth"      : <number> Range: 0.000–1.000.
    "pattern_boost"      : <number> Range: 0.000–0.350. The single highest
                                    pattern boost applied (0.000 if none).
  }

  "reasoning"           : <string>  2 to 4 sentences explaining the decision.
                                    Must name at least two specific field values
                                    from the input transaction and at least one
                                    rule_id or pattern_id. Must be honest about
                                    uncertainty if confidence < 0.70.

  "recommended_action"  : <string>  A complete, specific, actionable instruction
                                    that an analyst can execute immediately without
                                    re-reading the transaction record.
                                    Examples:
                                      "Approve — no action required."
                                      "Contact cardholder at phone number on file
                                       to verify the Lagos transaction before

                                      "Block card, initiate chargeback review,
                                       and flag linked device dev_shared_004 for
                                       cross-account investigation."

  }
